Quickstart

Hello, video — from your agent.

from widecast import WideCast

client = WideCast(api_key="wc_live_REPLACE_ME")

job = client.create_video(
    source="idea",
    idea_text="why founders should ship on day one",
    output_type="video",
)
print(client.status(job.id))  # poll /v1/status/{id}
import { WideCast } from "@widecast/sdk";

const client = new WideCast({ apiKey: "wc_live_REPLACE_ME" });

const job = await client.createVideo({
  source: "idea",
  ideaText: "why founders should ship on day one",
  outputType: "video",
});
await client.status(job.id); // poll /v1/status/{id}
{
  "tool": "create_video",
  "arguments": {
    "source": "idea",
    "idea_text": "why founders should ship on day one",
    "output_type": "video"
  }
}
# then poll status via tool: status(id)
curl -X POST https://api.widecast.ai/v1/create_video \
  -H "Authorization: Bearer wc_live_REPLACE_ME" \
  -H "Content-Type: application/json" \
  -d '{"source":"idea","idea_text":"why founders should ship on day one","output_type":"video"}'

curl https://api.widecast.ai/v1/status/{id} # poll

…then close the loop — create, (optionally) review, publish:

job = client.create_video(source="idea", idea_text="…").wait()
print(job.review_url)         # human-in-the-loop checkpoint

video = job.approve()
video.publish(accounts=["tiktok", "youtube", "instagram", "x", "linkedin"])
