Quickstart

Hello, video — from your agent.

from widecast import WideCast

wc = WideCast(api_key="wc_live_REPLACE_ME")

job = wc.create_video(
    source="idea",
    idea_text="why founders should ship on day one",
    output_type="video",
)

while job.status not in ("done", "error"):
    job = wc.status(job.id)   # GET /v1/status/{id}

print(job.review_url)
import { WideCast } from "@widecast/sdk";

const wc = new WideCast({ apiKey: "wc_live_REPLACE_ME" });

let job = await wc.createVideo({
  source: "idea",
  ideaText: "why founders should ship on day one",
  outputType: "video",
});

while (!["done", "error"].includes(job.status)) {
  job = await wc.status(job.id); // GET /v1/status/{id}
}

console.log(job.reviewUrl);
// agent invokes the MCP tool
{
  "tool": "widecast.create_video",
  "arguments": {
    "source": "idea",
    "idea_text": "why founders should ship on day one",
    "output_type": "video"
  }
}
// then poll widecast.status { "id": "<job_id>" }
curl -X POST https://api.widecast.ai/v1/create_video \
  -H "Authorization: Bearer wc_live_REPLACE_ME" \
  -H "Content-Type: application/json" \
  -d '{
    "source": "idea",
    "idea_text": "why founders should ship on day one",
    "output_type": "video"
  }'

# poll: curl https://api.widecast.ai/v1/status/{id}
